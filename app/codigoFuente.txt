/**
 * BaseController provides a convenient place for loading components
 * and performing functions that are needed by all your controllers.
 *
 * Extend this class in any new controllers:
 * ```
 *     class Home extends BaseController
 * ```
 *
 * For security, be sure to declare any new methods as protected or private.
 */
abstract class BaseController extends Controller
{
    protected User $user;
    protected array $roles_requeridos = [];
    /**
     * Be sure to declare properties for any property fetch you initialized.
     * The creation of dynamic property is deprecated in PHP 8.2.
     */

    // protected $session;

    /**
     * @return void
     */
    public function initController(RequestInterface $request, ResponseInterface $response, LoggerInterface $logger)
        {
            // Load here all helpers you want to be available in your controllers that extend BaseController.
            // Caution: Do not put the this below the parent::initController() call below.
            // $this->helpers = ['form', 'url'];

            // Caution: Do not edit this line.
            parent::initController($request, $response, $logger);
            $this->user = new User();
             if (!$this->user->tieneRol($this->roles_requeridos))
                {
                header('Location: ' . base_url('/'));
                exit();
                }

            // Preload any models, libraries, etc, here.
            // $this->session = service('session');
        }
    protected function idEventoFiltro(): ?int
        {
        return $this->user->tieneRol('Administrador')
            ? null
            : (int) $this->user->datos->id_evento;
        }

    protected function viewData(array $extra = []): array
        {
        return array_merge([
            'menu' => config('AppMenu')->menu,
            'user' => $this->user,
        ], $extra);
        }
}

<?php

namespace App\Controllers;

use App\Models\CalificacionesModel;

class Calificaciones extends BaseController
{
    protected array $roles_requeridos = ['Miembro Comité Científico'];

    public function getIndex()
        {
        $model      = new CalificacionesModel();
        $id_usuario = $this->user->datos->id_usuario;
        $id_evento  = $this->user->datos->id_evento;

        if ($id_evento === null)
            [$trabajos, $items] = $model->todosSinFiltroEvento($id_usuario);
        else
            [$trabajos, $items] = $model->trabajosConMisCalificaciones($id_evento, $id_usuario);

        return view('layout/main', $this->viewData([
            'active'       => 'calificaciones',
            'content_view' => 'calificaciones/listar',
            'content_data' => [
                'trabajos' => $trabajos,
                'items'    => $items,
            ],
        ]));
        }
    public function getDescargar(int $id_trabajo)
        {
        $db        = \Config\Database::connect();
        $id_evento = $this->user->datos->id_evento;

        // Verificar que el trabajo pertenece al evento del usuario
        $trabajo = $db->table('trabajos t')
            ->join('categorias c', 'c.id_categoria = t.id_categoria')
            ->where('t.id_trabajo', $id_trabajo)
            ->where('c.id_evento',  $id_evento)
            ->get()->getRow();

        if (!$trabajo)
            {
            header('Location: ' . base_url('/'));
            exit();
            }

        $row = $db->query(
            'SELECT pdf, nombre FROM trabajos WHERE id_trabajo = ?', [$id_trabajo]
        )->getRow();

        if (!$row || empty($row->pdf))
            {
            header('Location: ' . base_url('/'));
            exit();
            }

        $binario = pg_unescape_bytea($row->pdf);

        header('Content-Type: application/pdf');
        header('Content-Disposition: attachment; filename="' . $trabajo->nombre . '.pdf"');
        header('Content-Length: ' . strlen($binario));
        header('Cache-Control: no-cache');
        echo $binario;
        exit();
        }


    public function postGuardar()
        {
        $db         = \Config\Database::connect();
        $id_usuario = $this->user->datos->id_usuario;
        $id_evento  = $this->user->datos->id_evento;
        $id_trabajo = (int) $this->request->getPost('id_trabajo');
        $notas      = $this->request->getPost('notas');

        if (!$id_trabajo || !is_array($notas))
            {
            header('Location: ' . base_url('calificaciones') . '?error=1');
            exit();
            }

        $valido = $db->table('trabajos t')
            ->join('categorias c', 'c.id_categoria = t.id_categoria')
            ->where('t.id_trabajo', $id_trabajo)
            ->where('c.id_evento',  $id_evento)
            ->countAllResults();
        if (!$valido)
            {
            header('Location: ' . base_url('calificaciones') . '?error=1');
            exit();
            }

        $model = new CalificacionesModel();
        $model->guardarCalificaciones($id_usuario, $id_trabajo, $notas);

        header('Location: ' . base_url('calificaciones') . '?exito=1');
        exit();
        }
    public function postToggleAdmitido(): void
        {
            $idtrabajo = (int) $this->request->getPost('idtrabajo');

            if (!$idtrabajo)
            {
                $this->response->setStatusCode(400);
                echo json_encode(['ok' => false, 'error' => 'ID no válido']);
                exit;
            }

            // Verificar que el trabajo pertenece al evento del usuario
            $idevento = $this->user->datos->idevento;
            if ($idevento !== null)
            {
                $db      = \Config\Database::connect();
                $valido  = $db->table('trabajos t')
                    ->join('categorias c', 'c.idcategoria = t.idcategoria')
                    ->where('t.idtrabajo', $idtrabajo)
                    ->where('c.idevento', $idevento)
                    ->countAllResults();

                if (!$valido)
                {
                    $this->response->setStatusCode(403);
                    echo json_encode(['ok' => false, 'error' => 'No autorizado']);
                    exit;
                }
            }

            $model = new \App\Models\CalificacionesModel();
            $model->toggleAdmitido($idtrabajo);

            // Devolver el nuevo estado
            $db      = \Config\Database::connect();
            $trabajo = $db->table('trabajos')->select('admitido')->where('idtrabajo', $idtrabajo)->get()->getRow();

            echo json_encode(['ok' => true, 'admitido' => (bool) $trabajo->admitido]);
            exit;
        }




}

<?php

namespace App\Controllers;

use App\Models\UsuariosModel;

class Login extends \CodeIgniter\Controller
{
    public function getIndex()
        {
        if (session()->get('id_usuario'))
            {
            return redirect()->to(base_url('/'));
            }

        return view('login/index');
        }

    public function postIndex()
        {
        $login    = trim((string) $this->request->getPost('login'));
        $password = md5((string) $this->request->getPost('password'));

        $model = new UsuariosModel();

        $usuario = $model->where('login', $login)
                         ->where('password', $password)
                         ->first();

        if (!$usuario)
            {
            return view('login/index', [
                'error' => 'Usuario o contraseña incorrectos.',
            ]);
            }

        session()->set([
            'id_usuario' => $usuario->id_usuario,
            'login'      => $usuario->login,
            'nombre'     => $usuario->nombre,
        ]);

        return redirect()->to(base_url('/'));
        }

    public function getForgot()
        {
        return view('login/forgot');
        }

    public function postForgot()
        {
        $email = trim((string) $this->request->getPost('email'));
        $model = new UsuariosModel();
        $user  = $model->buscarPorEmail($email);

        if ($user)
            {
            $token  = bin2hex(random_bytes(32));
            $expira = date('Y-m-d H:i:s', strtotime('+1 hour'));

            $model->guardarTokenReset((int) $user->id_usuario, $token, $expira);

            $enlace = base_url('login/reset?token=' . urlencode($token));
            $this->enviarMailRecuperacion($user->email, $user->nombre, $enlace);
            }

        return view('login/forgot_ok');
        }

    public function getReset()
        {
        $token = (string) $this->request->getGet('token');
        $model = new UsuariosModel();
        $user  = $model->buscarPorResetTokenValido($token);

        if (!$user)
            {
            return view('login/reset_invalido');
            }

        return view('login/reset', [
            'token' => $token,
        ]);
        }

    public function postReset()
        {
        $token           = trim((string) $this->request->getPost('token'));
        $password        = (string) $this->request->getPost('password');
        $passwordConfirm = (string) $this->request->getPost('password_confirm');

        if ($password === '' || $password !== $passwordConfirm)
            {
            return view('login/reset', [
                'token' => $token,
                'error' => 'Las contraseñas no coinciden.',
            ]);
            }

        $model = new UsuariosModel();
        $user  = $model->buscarPorResetTokenValido($token);

        if (!$user)
            {
            return view('login/reset_invalido');
            }

        $model->actualizarPasswordConReset((int) $user->id_usuario, md5($password));

        return view('login/reset_ok');
        }

    public function getLogout()
        {
        session()->destroy();
        return redirect()->to(base_url('login'));
        }

    private function enviarMailRecuperacion(string $email, string $nombre, string $enlace): void
        {
        $mailer = \Config\Services::email();

        $mailer->setFrom('noreply@almansa.ovh', 'JIGA 2');
        $mailer->setTo($email);
        $mailer->setSubject('Recuperación de contraseña');
        $mailer->setMessage(view('emails/recuperar_password', [
            'nombre' => $nombre,
            'enlace' => $enlace,
        ]));

        if (!$mailer->send())
            {
            log_message('error', 'Error enviando recuperación a ' . $email . ': ' . $mailer->printDebugger(['headers']));
            }
        }
}

    public function postCrear()
        {
        $model     = new PlantillasModel();
        $nombre    = $this->request->getPost('nombre');
        $id_evento = (int) $this->request->getPost('id_evento');
        $privada   = $this->request->getPost('privada') ? true : false;
        $archivo   = $this->request->getFile('archivo');

        if ($archivo && $archivo->getError() === UPLOAD_ERR_INI_SIZE)
            {
            return redirect()->back()->with('error', 'El archivo es demasiado grande.');
            }

        if ($archivo && $archivo->isValid() && $archivo->getSize() > 0)
            {
            $ok = $model->crearConArchivo(
                $nombre,
                $archivo->getClientName(),
                $id_evento,
                $privada,
                file_get_contents($archivo->getTempName())
            );
            }
        else
            {
            $ok = $model->crearSinArchivo($nombre, $id_evento, $privada);
            }

        if ($ok === false)
            {
            header('Location: ' . base_url('plantillas/listar'));
            exit();
            }

        header('Location: ' . base_url('plantillas/listar'));
        exit();
        }

<?php

namespace App\Controllers;

use App\Models\UsuariosModel;

class Registro extends \CodeIgniter\Controller
{
    // ----------------------------------------------------------------
    // GET: muestra el formulario de registro
    // ----------------------------------------------------------------
    public function getIndex()
        {
        $id_evento = (int) $this->request->getGet('evento');

        return view('registro/formulario', [
            'id_evento' => $id_evento,
        ]);
        }

    // ----------------------------------------------------------------
    // POST: procesa el registro
    // ----------------------------------------------------------------
    public function postIndex()
        {
        $model     = new UsuariosModel();
        $id_evento = (int) $this->request->getPost('id_evento');
        $email     = trim($this->request->getPost('email'));
        $nombre    = trim($this->request->getPost('nombre'));
        $password  = $this->request->getPost('password');
        $nombre_evento = '';
        
        // Verificar que el email no existe ya
        if ($model->emailExiste($email))
            {
            return redirect()->back()
                             ->with('error', 'Ya existe una cuenta con ese email.');
            }

        $token = bin2hex(random_bytes(32));

        $datos = (object) [
            'login'               => $email,
            'email'               => $email,
            'nombre'              => $nombre,
            'password'            => md5($password),
            'id_evento'           => $id_evento ?: null,
            'validado'            => false,
            'correo_verificado'   => false,
            'token_verificacion'  => $token,
        ];

        if ($model->crear($datos) === false)
            {
            return redirect()->back()
                             ->with('error', 'Error al crear la cuenta.');
            }

        if ($id_evento)
            {
            $row = (new \App\Models\EventosModel())
                ->where('id_evento', $id_evento)
                ->get()
                ->getRow();
            $nombre_evento = $row->nombre ?? '';
            }

        $enlace = base_url('registro/verificar?token=' . $token);
        $this->enviarMailVerificacion($email, $nombre, $enlace, $nombre_evento);


        return view('registro/pendiente', [
            'email' => $email,
        ]);
        }

    // ----------------------------------------------------------------
    // GET: verificación del token
    // ----------------------------------------------------------------
    public function getVerificar()
        {
        $token = $this->request->getGet('token');
        $model = new UsuariosModel();

        $usuario = $model->porToken($token);

        if (!$usuario)
            {
            return view('registro/token_invalido');
            }

        $model->verificarCorreo($usuario->id_usuario);

        return view('registro/verificado');
        }

    // ----------------------------------------------------------------
    // Stub de envío de correo 
    // ----------------------------------------------------------------
    private function enviarMailVerificacion(
        string $email,
        string $nombre,
        string $enlace,
        string $nombre_evento = ''
     ): void
        {
        $mailer = \Config\Services::email();

        $mailer->setFrom('noreply@almansa.ovh', 'JIGA 2');
        $mailer->setTo($email);
        $mailer->setSubject('Verifica tu cuenta en ' . ($nombre_evento ?: 'JIGA 2'));
        $mailer->setMessage(view('emails/verificacion', [
            'nombre'        => $nombre,
            'enlace'        => $enlace,
            'nombre_evento' => $nombre_evento ?: 'JIGA 2',
        ]));

        if (!$mailer->send())
            {
            log_message('error', 'Error enviando verificación a ' . $email . ': ' . $mailer->printDebugger(['headers']));
            }
        }

}

        ]));

        if (!$mailer->send())
            {
            log_message('error', 'Error enviando notif. comité a ' . $miembro->email);
            }
        }


    private function procesarPdf(?object $archivo): ?array
        {
        if (!$archivo || !$archivo->isValid() || $archivo->getSize() === 0)
            {
            return null;
            }
        if ($archivo->getError() === UPLOAD_ERR_INI_SIZE)
            {
            throw new \RuntimeException('grande');
            }
        if (strtolower($archivo->getClientExtension()) !== 'pdf')
            {
            throw new \RuntimeException('formato');
            }
        return [
            'filename' => $archivo->getClientName(),
            'binario'  => file_get_contents($archivo->getTempName()),
        ];
        }

    public function postCrear()
        {
        $model         = new TrabajosModel();
        $nombre        = trim($this->request->getPost('nombre'));
        $lista_autores = trim($this->request->getPost('lista_autores'));
        $id_categoria  = (int) $this->request->getPost('id_categoria');
        $id_usuario    = $this->user->datos->id_usuario;

        try
            {
            $pdf = $this->procesarPdf($this->request->getFile('pdf'));
            }
        catch (\RuntimeException $e)
            {
            $msg = $e->getMessage() === 'grande'
                ? 'El archivo es demasiado grande.'
                : 'Solo se admiten archivos PDF.';
            header('Location: ' . base_url('/'));
            exit();
            }

        if (!$pdf)
            {
            header('Location: ' . base_url('/'));
            exit();
            }

        $ok          = $model->crearConPdf(
            $nombre, $lista_autores, $id_categoria,
            $id_usuario, $pdf['filename'], $pdf['binario']
        );
        $filename_pdf = $pdf['filename'];
        unset($pdf);

        if ($ok === false)
            {
            header('Location: ' . base_url('/') . '?error=1');
            exit();
            }

        $this->enviarMailPresentador(
            $this->user->datos->email,
            $this->user->datos->nombre,
            $ok,
            $nombre,
            $lista_autores,
            $filename_pdf
        );

        $miembros = $model->miembrosComiteCientifico($this->user->datos->id_evento);
        // DEBUG TEMPORAL
        log_message('debug', 'Miembros comite: ' . print_r($miembros, true));
        foreach ($miembros as $miembro)
            {
            $this->enviarMailComiteCientifico($miembro, $nombre);
            }

        header('Location: ' . base_url('/') . '?exito=1');
        exit();
        }


    public function postActualizar()
        {
        $model         = new TrabajosModel();
        $id            = (int) $this->request->getPost('id_trabajo');
        $nombre        = trim($this->request->getPost('nombre'));

        $trabajo = $model->find($id);
        if (!$trabajo || $trabajo->id_usuario != $this->user->datos->id_usuario)

            exit();

        try

        $id_usuario = (int) $this->request->getPost('id_usuario');

        if (empty($id_usuario))
            {
            return redirect()->back()->with('error', 'ID no válido.');
            }

        $usuario = $model->find($id_usuario);

        if (!$usuario)
            {
            return redirect()->back()->with('error', 'Usuario no encontrado.');
            }

        // Marcar como validado y asignar rol
        $model->validarUsuario($id_usuario);

        // Enviar email de bienvenida
        $this->enviarMailAdmitido(
            $usuario->email,
            $usuario->nombre,
            base_url('login')
        );

        return redirect()->to(base_url('validacion/listar'))
                         ->with('exito', 'Usuario validado correctamente.');
        }

    public function postEliminar()
        {
        $model      = new UsuariosModel();
        $id_usuario = (int) $this->request->getPost('id_usuario');

        if (empty($id_usuario))
            {
            return redirect()->back()->with('error', 'ID no válido.');
            }

        if ($model->borrar($id_usuario) === false)
            {
            return redirect()->back()->with('error', 'Error al eliminar el usuario.');
            }

        return redirect()->to(base_url('validacion/listar'))
                         ->with('exito', 'Usuario eliminado correctamente.');
        }

    private function enviarMailAdmitido(

class CalificacionesModel extends Model

    protected $table         = 'calificaciones';

    protected $allowedFields = ['id_usuario', 'id_trabajo', 'id_item', 'nota'];

    /**
     * Trabajos del evento con las notas del calificador (solo las suyas).
     * El evento se deduce por usuarios.id_evento a través de id_usuario del trabajo.
     */
    public function trabajosConMisCalificaciones(int $id_evento, int $id_usuario_calificador): array
        {
        // Items del evento
        $items = $this->db->table('items')
            ->where('id_evento', $id_evento)
            ->get()->getResultObject();

        // Trabajos del evento (via categoria → id_evento)
        $trabajos = $this->db->table('trabajos t')
            ->select('t.id_trabajo, t.nombre, t.lista_autores, t.admitido, cat.nombre AS categoria')
            ->join('categorias cat', 'cat.id_categoria = t.id_categoria')
            ->where('cat.id_evento', $id_evento)
            ->get()->getResultObject();
        // Calificaciones del calificador para este evento
        $cals = $this->db->table('calificaciones c')
            ->select('c.id_trabajo, c.id_item, c.nota')
            ->join('items i', 'i.id_item = c.id_item')
            ->where('c.id_usuario', $id_usuario_calificador)
            ->where('i.id_evento', $id_evento)
            ->get()->getResultArray();

        // Indexar por [id_trabajo][id_item] => nota
        $index = [];
        foreach ($cals as $c)
            $index[$c['id_trabajo']][$c['id_item']] = $c['nota'];

        // Montar resultado con estado de tres valores
        $total = count($items);
        foreach ($trabajos as $t)
            {
            $t->items    = [];
            $con_nota    = 0;

            foreach ($items as $item)
                {
                $nota = $index[$t->id_trabajo][$item->id_item] ?? null;
                $t->items[] = (object)[
                    'id_item' => $item->id_item,
                    'nombre'  => $item->nombre,
                    'nota'    => $nota,
                ];
                if ($nota !== null) $con_nota++;
                }

            $t->estado = match(true)
                {
                $con_nota === 0      => 'pendiente',
                $con_nota === $total => 'completo',
                default              => 'parcial',
                };
            }

        return [$trabajos, $items];
        }


    public function guardarCalificaciones(int $id_usuario, int $id_trabajo, array $notas): void
        {
        foreach ($notas as $id_item => $nota)
            {
            // Si viene vacío, guardar null (o no insertar si no existe)
            $nota_valor = ($nota !== '' && $nota !== null)
                ? max(0, min(10, (float) $nota))
                : null;

            $existe = $this->db->table('calificaciones')
                ->where('id_usuario', $id_usuario)
                ->where('id_trabajo', $id_trabajo)
                ->where('id_item',    (int) $id_item)
                ->countAllResults();

            if ($existe)
                {
                $this->db->table('calificaciones')
                    ->where('id_usuario', $id_usuario)
                    ->where('id_trabajo', $id_trabajo)
                    ->where('id_item',    (int) $id_item)
                    ->update(['nota' => $nota_valor]);
                }
            else if ($nota_valor !== null)
                {
                // Solo insertar si hay nota — no crear registros vacíos
                $this->db->table('calificaciones')->insert([
                    'id_usuario' => $id_usuario,
                    'id_trabajo' => $id_trabajo,
                    'id_item'    => (int) $id_item,
                    'nota'       => $nota_valor,
                ]);
                }
            }
        }

<?php

namespace App\Models;

use CodeIgniter\Model;

class CrudModel extends Model
{
    protected $useTimestamps = false;
    protected $returnType    = 'object';

    // Propiedad genérica para el registro seleccionado
    public ?object $Registro = null;

    // ----------------------------------------------------------------
    // CONSTRUCTOR
    // Si se proporciona un id, carga el registro en $this->Registro.
    // ----------------------------------------------------------------
    public function __construct(?int $id = null)
        {
        parent::__construct();

        if ($id !== null)
            {
            $this->Registro = $this->find($id) ?? null;
            }
        }

    // ----------------------------------------------------------------
    // CREAR
    // ----------------------------------------------------------------
    public function crear(object $datos): int|false
        {
        $arr = (array) $datos;
        $arr = array_intersect_key($arr, array_flip($this->allowedFields));

        if ($this->insert($arr))
            {
            return (int) $this->getInsertID();
            }

        return false;
        }

    // ----------------------------------------------------------------
    // BORRAR
    // ----------------------------------------------------------------
    public function borrar(int $id): bool
        {
        return (bool) $this->delete($id);
        }

    // ----------------------------------------------------------------
    // ACTUALIZAR
    // ----------------------------------------------------------------
    public function actualizar(object $datos): bool
        {
        $arr = (array) $datos;

        if (empty($arr[$this->primaryKey]))
            {
            return false;
            }

        $id = (int) $arr[$this->primaryKey];
        unset($arr[$this->primaryKey]);

        $actualizar = array_intersect_key($arr, array_flip($this->allowedFields));

        if (empty($actualizar))
            {
            return true;
            }

        if ($this->find($id) === null)
            {
            return false;
            }

        return (bool) $this->update($id, $actualizar);
        }

    // ----------------------------------------------------------------
    // GUARDAR
    // Persiste $this->Registro en BD.
    // ----------------------------------------------------------------
    public function guardar(): bool
        {
        if ($this->Registro === null)
            {
            return false;
            }

        return $this->actualizar($this->Registro);
        }

    // ----------------------------------------------------------------
    // LISTAR
    // ----------------------------------------------------------------
    public function listar(): array
        {
        return $this->orderBy($this->primaryKey, 'ASC')->findAll();
        }
    }

{
    private ?int $id_usuario = null;
    public object  $datos;
    public array   $roles = [];

    public function __construct()
        {
        parent::__construct();

        $this->id_usuario = session()->get('id_usuario');

        if (!$this->id_usuario)
            {
            header('Location: ' . base_url('login'));
            exit();
            }

        // Datos del usuario
        $this->datos = $this->db
            ->table('usuarios u')
            ->select('u.*, e.nombre AS nombre_evento')
            ->join('eventos e', 'e.id_evento = u.id_evento', 'left')
            ->where('u.id_usuario', $this->id_usuario)
            ->get()
            ->getRow();

        // Roles del usuario
        $filas = $this->db
            ->table('rel_usuarios_roles ur')
            ->select('r.nombre')
            ->join('roles r', 'r.id_rol = ur.id_rol')
            ->where('ur.id_usuario', $this->id_usuario)
            ->get()
            ->getResult();

        $this->roles = array_column(
            array_map(fn($f) => (array)$f, $filas),
            'nombre'
        );
        }

    public function tieneRol(string|array $roles = []): bool
        {
        if (empty($roles))      // ¿Seguro? Creo que debe ser false.
            return true;

        if (in_array('Administrador', $this->roles))
            return true;            // los administradores siempre tienen acceso

        $roles = (array)$roles;
        foreach ($roles as $rol)
            {
            if (in_array($rol, $this->roles))
                return true;
            }
        return false;
        }


}

        if ($id_evento !== null)
            {
            $builder->where('u.id_evento', $id_evento);
            }

        $usuarios = $builder->get()->getResultObject();

        foreach ($usuarios as $usuario)
            {
            $usuario->roles = $this->db
                ->table('rel_usuarios_roles r')
                ->select('r.id_rol, ro.nombre')
                ->join('roles ro', 'ro.id_rol = r.id_rol')
                ->where('r.id_usuario', $usuario->id_usuario)
                ->orderBy('ro.nombre', 'ASC')
                ->get()
                ->getResultObject();
            }

        return $usuarios;
        }

    public function eventosDisponibles(): array
        {
        return $this->db
            ->table('eventos')
            ->orderBy('nombre', 'ASC')
            ->get()
            ->getResultObject();
        }

    public function rolesDisponibles(int $id_usuario): array
        {
        $asignados = $this->db
            ->table('rel_usuarios_roles')
            ->select('id_rol')
            ->where('id_usuario', $id_usuario)
            ->get()
            ->getResultArray();

        $ids_asignados = array_column($asignados, 'id_rol');

        $query = $this->db->table('roles')->orderBy('nombre', 'ASC');

        if (!empty($ids_asignados))
            {
            $query->whereNotIn('id_rol', $ids_asignados);
            }

        return $query->get()->getResultObject();
        }

    public function asignarRol(int $id_usuario, int $id_rol): bool
        {
        $existe = $this->db
            ->table('rel_usuarios_roles')
            ->where('id_usuario', $id_usuario)
            ->where('id_rol', $id_rol)
            ->countAllResults();

        if ($existe) { return false; }

        return (bool) $this->db
            ->table('rel_usuarios_roles')
            ->insert(['id_usuario' => $id_usuario, 'id_rol' => $id_rol]);
        }

            ->where('u.validado', false)
            ->orderBy('u.id_usuario', 'ASC');

        if ($id_evento !== null)

            $builder->where('u.id_evento', $id_evento);

        $this->db->table('usuarios')
            ->where('id_usuario', $id_usuario)
            ->update(['validado' => true]);

        // Buscar id del rol 'Presentador de Trabajo'
        $rol = $this->db->table('roles')
            ->where('nombre', 'Presentador de Trabajo')
            ->get()
            ->getRow();

        if (!$rol) { return; }

        // Asignar rol si no lo tiene ya
        $existe = $this->db->table('rel_usuarios_roles')
            ->where('id_usuario', $id_usuario)
            ->where('id_rol', $rol->id_rol)
            ->countAllResults();

        if (!$existe)
            {
            $this->db->table('rel_usuarios_roles')->insert([

<?php

use CodeIgniter\CLI\CLI;

// The main Exception
CLI::write('[' . $exception::class . ']', 'light_gray', 'red');
CLI::write($message);
CLI::write('at ' . CLI::color(clean_path($exception->getFile()) . ':' . $exception->getLine(), 'green'));
CLI::newLine();

$last = $exception;

while ($prevException = $last->getPrevious()) {
    $last = $prevException;

    CLI::write('  Caused by:');
    CLI::write('  [' . $prevException::class . ']', 'red');
    CLI::write('  ' . $prevException->getMessage());
    CLI::write('  at ' . CLI::color(clean_path($prevException->getFile()) . ':' . $prevException->getLine(), 'green'));
    CLI::newLine();
}

// The backtrace
if (defined('SHOW_DEBUG_BACKTRACE') && SHOW_DEBUG_BACKTRACE) {
    $backtraces = $last->getTrace();

    if ($backtraces) {
        CLI::write('Backtrace:', 'green');
    }

    foreach ($backtraces as $i => $error) {
        $padFile  = '    '; // 4 spaces
        $padClass = '       '; // 7 spaces
        $c        = str_pad($i + 1, 3, ' ', STR_PAD_LEFT);

        if (isset($error['file'])) {
            $filepath = clean_path($error['file']) . ':' . $error['line'];

            CLI::write($c . $padFile . CLI::color($filepath, 'yellow'));
        } else {
            CLI::write($c . $padFile . CLI::color('[internal function]', 'yellow'));
        }

        $function = '';

        if (isset($error['class'])) {
            $type = ($error['type'] === '->') ? '()' . $error['type'] : $error['type'];
            $function .= $padClass . $error['class'] . $type . $error['function'];
        } elseif (! isset($error['class']) && isset($error['function'])) {
            $function .= $padClass . $error['function'];
        }

        $args = implode(', ', array_map(static fn ($value): string => match (true) {
            is_object($value) => 'Object(' . $value::class . ')',
            is_array($value)  => $value !== [] ? '[...]' : '[]',
            $value === null   => 'null', // return the lowercased version
            default           => var_export($value, true),
        }, array_values($error['args'] ?? [])));

        $function .= '(' . $args . ')';

        CLI::write($function);
        CLI::newLine();
    }
}

var tabLinks    = new Array();
var contentDivs = new Array();

function init()
{
    // Grab the tab links and content divs from the page
    var tabListItems = document.getElementById('tabs').childNodes;
    console.log(tabListItems);
    for (var i = 0; i < tabListItems.length; i ++)
    {
        if (tabListItems[i].nodeName == "LI")
        {
            var tabLink     = getFirstChildWithTagName(tabListItems[i], 'A');
            var id          = getHash(tabLink.getAttribute('href'));
            tabLinks[id]    = tabLink;
            contentDivs[id] = document.getElementById(id);
        }
    }

    // Assign onclick events to the tab links, and
    // highlight the first tab
    var i = 0;

    for (var id in tabLinks)
    {
        tabLinks[id].onclick = showTab;
        tabLinks[id].onfocus = function () {
            this.blur()
        };
        if (i == 0)
        {
            tabLinks[id].className = 'active';
        }
        i ++;
    }

    // Hide all content divs except the first
    var i = 0;

    for (var id in contentDivs)
    {
        if (i != 0)
        {
            console.log(contentDivs[id]);
            contentDivs[id].className = 'content hide';
        }
        i ++;
    }
}

function showTab()
{
    var selectedId = getHash(this.getAttribute('href'));

    // Highlight the selected tab, and dim all others.
    // Also show the selected content div, and hide all others.
    for (var id in contentDivs)
    {
        if (id == selectedId)
        {
            tabLinks[id].className    = 'active';
            contentDivs[id].className = 'content';
        }
        else
        {
            tabLinks[id].className    = '';
            contentDivs[id].className = 'content hide';
        }
    }

    // Stop the browser following the link
    return false;
}

function getFirstChildWithTagName(element, tagName)
{
    for (var i = 0; i < element.childNodes.length; i ++)
    {
        if (element.childNodes[i].nodeName == tagName)
        {
            return element.childNodes[i];
        }
    }
}

function getHash(url)
{
    var hashPos = url.lastIndexOf('#');
    return url.substring(hashPos + 1);
}

function toggle(elem)
{
    elem = document.getElementById(elem);

    if (elem.style && elem.style['display'])
    {
        // Only works with the "style" attr
        var disp = elem.style['display'];
    }
    else if (elem.currentStyle)
    {
        // For MSIE, naturally
        var disp = elem.currentStyle['display'];
    }
    else if (window.getComputedStyle)
    {
        // For most other browsers
        var disp = document.defaultView.getComputedStyle(elem, null).getPropertyValue('display');
    }

    // Toggle the state of the "display" style
    elem.style.display = disp == 'block' ? 'none' : 'block';

    return false;
}

function icono_extension(string $filename): string
    {
    $ext = strtolower(pathinfo($filename, PATHINFO_EXTENSION));
    $iconos = [
        'pdf'  => '📕',
        'doc'  => '📘',
        'docx' => '📘',
        'ppt'  => '📙',
        'pptx' => '📙',
        'xls'  => '📗',
        'xlsx' => '📗',
    ];
    $emoji = $iconos[$ext] ?? '📄';
    return '<span style="font-size:48px;line-height:1">' . $emoji . '</span>';
    }

<script>
// Columnas y configuración pasadas desde PHP
const CRUD_COLUMNAS  = <?= json_encode($columnas) ?>;
const CRUD_PK        = <?= json_encode($pk) ?>;
const CRUD_RUTA_BASE = <?= json_encode($ruta_base) ?>;
const CRUD_CAMPOS_FORM = <?= json_encode($campos_form ?? null) ?>;
const CRUD_ENCTYPE      = <?= json_encode($enctype      ?? null) ?>;
const CRUD_COL_DESCARGA = <?= json_encode($col_descarga ?? null) ?>;
const CRUD_RUTA_DESCARGA= <?= json_encode($ruta_descarga ?? null) ?>;
const CRUD_ACCIONES_EXTRA = <?= json_encode($acciones_extra ?? []) ?>;




$(function ()
    {
    const overlay   = $('#crud-modal-overlay');
    const form      = $('#crud-form');
    const modalBody = $('#modal-fields');

    // ── Genera los campos del formulario a partir de $columnas ──
function generarCampos(datos)
    {
    modalBody.empty();

    // Si hay campos_form personalizados, los usamos; si no, generamos desde columnas
    const fuente = CRUD_CAMPOS_FORM ?? CRUD_COLUMNAS.map(c => ({
        campo : c.campo,
        label : c.label,
        tipo  : 'text',
    }));

    fuente.forEach(function (col)
        {
        if (col.campo.toLowerCase().startsWith('id') && !CRUD_CAMPOS_FORM) { return; }

        const valor = datos ? (datos[col.campo] ?? '') : (col.valor ?? '');

        if (col.tipo === 'hidden')
            {
            modalBody.append(
                `<input type="hidden" name="${col.campo}" value="${valor}">`
            );
            return;
            }

        if (col.tipo === 'select')
            {
            let options = col.opciones.map(function (opt)
                {
                const selected = String(opt[col.opt_val]) === String(valor) ? 'selected' : '';
                return `<option value="${opt[col.opt_val]}" ${selected}>${opt[col.opt_txt]}</option>`;
                }).join('');

            modalBody.append(`
                <label>
                    ${col.label}
                    <select name="${col.campo}" required>${options}</select>
                </label>`);
            return;
            }

        if (col.tipo === 'number')
            {
            const step = col.step ?? 'any';
            modalBody.append(`
                <label>
                    ${col.label}
                    <input type="number"
                        step="${step}"
                        name="${col.campo}"
                        value="${$('<div>').text(valor).html()}">
                </label>`);
            return;
            }
        if (col.tipo === 'file')
            {
            const accept = col.accept ?? '';
            const nota   = col.nota   ?? '';
            modalBody.append(`
                <label>
                    ${col.label}
                    ${nota ? `<small style="color:#888">${nota}</small>` : ''}
                    <input type="file"
                        name="${col.campo}"
                        accept="${accept}">
                </label>`);
            return;
            }
        if (col.tipo === 'checkbox')
            {
            const checked = valor == true || valor === 't' || valor === '1' ? 'checked' : '';
            modalBody.append(`
                <label style="flex-direction:row; align-items:center; gap:10px; cursor:pointer;">
                    <input type="checkbox"
                        name="${col.campo}"
                        value="1"
                        ${checked}
                        style="width:auto; margin:0;">
                    ${col.label}
                </label>`);
            return;
            }

        // Tipo text por defecto
        modalBody.append(`
            <label>
                ${col.label}
                <input type="text"
                       name="${col.campo}"
                       value="${$('<div>').text(valor).html()}">
            </label>`);
        });
    }


    // ── Abre el modal ──
    function abrirModal(titulo, datos, id)
        {
        $('#modal-titulo').text(titulo);
        $('#form-pk').val(id ?? '');
        generarCampos(datos);

        const accion = id
            ? `${CRUD_RUTA_BASE}/actualizar`
            : `${CRUD_RUTA_BASE}/crear`;

        form.attr('action', accion);

        // enctype para subida de archivos
        if (CRUD_ENCTYPE)
            {
            form.attr('enctype', CRUD_ENCTYPE);
            }
        else
            {
            form.removeAttr('enctype');
            }

        overlay.fadeIn(150);
        modalBody.find('input').first().focus();
    }


    // ── Cierra el modal ──
    function cerrarModal()
        {
        overlay.fadeOut(150);
        form[0].reset();
        }

    // ── Click en "Nuevo" ──
    $('#btn-nuevo').on('click', function ()
        {
        abrirModal('Nuevo registro', null, null);
        });

    // ── Click en fila → editar ──
    $('#crud-tabla').on('click', 'tr.crud-fila', function (e)
        {
        // Evitamos que el click en los botones de acción abra el modal dos veces
        if ($(e.target).closest('.btn-editar, .btn-borrar').length) { return; }

        const datos = JSON.parse($(this).attr('data-fila'));
        const id    = $(this).attr('data-id');

    // ── Click en botón "Editar" ──
    $('#crud-tabla').on('click', '.btn-editar', function (e)

        e.stopPropagation();

        abrirModal('Editar registro', datos, id);
        });

    // ── Click en botón "Borrar" ──
    $('#crud-tabla').on('click', '.btn-borrar', function (e)

        e.stopPropagation();

                location.reload();
                })
            .fail(function ()
                {
                alert('Error al borrar el registro.');
                });
        });

    // ── Cerrar modal ──
    $('#modal-cerrar, #btn-cancelar').on('click', cerrarModal);

    overlay.on('click', function (e)
        {
        if ($(e.target).is(overlay)) { cerrarModal(); }
        });

<script>
pdfjsLib.GlobalWorkerOptions.workerSrc =
    'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';

const PDF_URL    = '<?= base_url('plantillas/descargar/' . $plantilla->id_plantilla) ?>';
let   pdfPageW   = 595;  // puntos PDF (A4)
let   pdfPageH   = 842;
let   canvasW    = 0;
let   canvasH    = 0;

// Cargar y renderizar PDF
pdfjsLib.getDocument(PDF_URL).promise.then(function(pdf)
    {
    return pdf.getPage(1);
    })
    .then(function(page)
        {
        const viewport = page.getViewport({ scale: 1.5 });
        const canvas   = document.getElementById('pdf-canvas');
        const ctx      = canvas.getContext('2d');

        canvas.width  = viewport.width;
        canvas.height = viewport.height;
        canvasW = viewport.width;
        canvasH = viewport.height;

        // Dimensiones reales del PDF en puntos
        pdfPageW = page.getViewport({ scale: 1 }).width;
        pdfPageH = page.getViewport({ scale: 1 }).height;

        page.render({ canvasContext: ctx, viewport: viewport });

        dibujarCuadricula();
        colocarMarcadoresIniciales();
        });

function dibujarCuadricula()
    {
    const grid   = document.getElementById('grid-canvas');
    grid.width   = canvasW;
    grid.height  = canvasH;
    const ctx    = grid.getContext('2d');
    const paso   = 50; // puntos PDF entre líneas

    ctx.strokeStyle = 'rgba(100,149,237,0.25)';
    ctx.lineWidth   = 0.5;
    ctx.font        = '9px Arial';
    ctx.fillStyle   = 'rgba(100,149,237,0.6)';

    // Líneas verticales
    for (let x = 0; x <= pdfPageW; x += paso)
        {
        const cx = pdfToCanvas(x, 0).cx;
        ctx.beginPath();
        ctx.moveTo(cx, 0);
        ctx.lineTo(cx, canvasH);
        ctx.stroke();
        if (x > 0) ctx.fillText(x, cx + 2, 10);
        }

    // Líneas horizontales
    for (let y = 0; y <= pdfPageH; y += paso)
        {
        const cy = pdfToCanvas(0, y).cy;
        ctx.beginPath();
        ctx.moveTo(0, cy);
        ctx.lineTo(canvasW, cy);
        ctx.stroke();
        if (y > 0) ctx.fillText(y, 2, cy - 2);
        }
    }

// Convierte coordenadas PDF → canvas
function pdfToCanvas(px, py)
    {
    return {
        cx: (px / pdfPageW) * canvasW,
        cy: (py / pdfPageH) * canvasH,
    };
    }

// Convierte coordenadas canvas → PDF
function canvasToPdf(cx, cy)
    {
    return {
        px: (cx / canvasW) * pdfPageW,
        py: (cy / canvasH) * pdfPageH,
    };
    }

function colocarMarcadoresIniciales()
    {
    moverMarcador('titulo',
        parseFloat(document.getElementById('pos_x_titulo').value),
        parseFloat(document.getElementById('pos_y_titulo').value));

    moverMarcador('autores',
        parseFloat(document.getElementById('pos_x_autores').value),
        parseFloat(document.getElementById('pos_y_autores').value));
    }

function moverMarcador(tipo, px, py)
    {
    const wrapper = document.getElementById('pdf-wrapper');
    const marker  = document.getElementById('marker-' + tipo);
    const rect    = wrapper.getBoundingClientRect();
    const pos     = pdfToCanvas(px, py);

    marker.style.left    = pos.cx + 'px';
    marker.style.top     = pos.cy + 'px';
    marker.style.display = 'block';
    }

// Click en el PDF
document.getElementById('pdf-canvas').addEventListener('click', function(e)
    {
    const rect   = this.getBoundingClientRect();
    const cx     = e.clientX - rect.left;
    const cy     = e.clientY - rect.top;
    const {px, py} = canvasToPdf(cx, cy);
    const tipo   = document.querySelector('input[name="marcador_activo"]:checked').value;

    if (tipo === 'titulo')
        {
        document.getElementById('pos_x_titulo').value = px.toFixed(1);
        document.getElementById('pos_y_titulo').value = py.toFixed(1);
        }
    else
        {
        document.getElementById('pos_x_autores').value = px.toFixed(1);
        document.getElementById('pos_y_autores').value = py.toFixed(1);
        }

    moverMarcador(tipo, px, py);
    });
</script>

<script>
function abrirModalEditar(id, login, nombre, id_evento)
    {
    document.getElementById('edit_id_usuario').value = id;
    document.getElementById('edit_login').value      = login;
    document.getElementById('edit_nombre').value     = nombre;
    document.getElementById('edit_password').value   = '';

    const sel = document.getElementById('edit_id_evento');
    if (sel) { sel.value = id_evento; }

    new bootstrap.Modal(document.getElementById('modalEditarUsuario')).show();
    }
function abrirModalRol(id_usuario, login)
    {
    document.getElementById('rol_id_usuario').value     = id_usuario;
    document.getElementById('rol_login_label').textContent = login;

    const select = document.getElementById('selectRoles');
    select.innerHTML = '<option value="">Cargando...</option>';

    fetch(`<?= base_url('usuarios/roles-disponibles') ?>/${id_usuario}`)
        .then(r => r.json())
        .then(roles => {
            if (roles.length === 0)
                {
                select.innerHTML =
                    '<option value="">No hay roles disponibles</option>';
                return;
                }
            select.innerHTML = roles.map(r =>
                `<option value="${r.id_rol}">${r.nombre}</option>`
            ).join('');
        });

    new bootstrap.Modal(document.getElementById('modalAsignarRol')).show();
    }
</script>
